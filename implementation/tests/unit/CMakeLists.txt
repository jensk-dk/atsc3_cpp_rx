# One Catch2 executable for all layers; run a subset with tags, e.g. atsc3_unit "[alp]".
file(GLOB_RECURSE ATSC3_UNIT_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/*.cpp)
add_executable(atsc3_unit ${ATSC3_UNIT_SOURCES})
target_include_directories(atsc3_unit PRIVATE ${PROJECT_SOURCE_DIR}/src ${PROJECT_SOURCE_DIR}/tuners/pcap)
target_link_libraries(atsc3_unit PRIVATE atsc3_test_support Catch2::Catch2WithMain)
if(ATSC3_WEB)
  target_link_libraries(atsc3_unit PRIVATE atsc3::web)
  target_compile_definitions(atsc3_unit PRIVATE ATSC3_HAVE_WEB=1)
endif()
if(ATSC3_VLC)
  target_link_libraries(atsc3_unit PRIVATE atsc3_vlc_mrl)
  target_compile_definitions(atsc3_unit PRIVATE ATSC3_HAVE_VLC_MRL=1)
endif()
if(ATSC3_STB)
  target_link_libraries(atsc3_unit PRIVATE atsc3_stb_model)
  target_compile_definitions(atsc3_unit PRIVATE ATSC3_HAVE_STB_MODEL=1)
endif()
atsc3_set_test_warnings(atsc3_unit)
catch_discover_tests(atsc3_unit PROPERTIES LABELS unit)
